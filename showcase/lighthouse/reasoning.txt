function beamMaterial(intensity){

    blending: THREE.AdditiveBlending,
    uniforms: {

      varying vec2 vUv; varying vec3 vN; varying vec3 vW;
      void main(){
        vUv = uv;
        vN = normalize(mat3(modelMatrix) * normal);
        vec4 wp = modelMatrix * vec4(position, 1.0);
        vW = wp.xyz;
        gl_Position = projectionMatrix * viewMatrix * wp;
      }`,
    fragmentShader: `
      uniform vec3 uColor; uniform float uIntensity; uniform vec3 uFogColor;
      varying vec2 vUv; varying vec3 vN; varying vec3 vW;
      void main(){
        vec3 V = normalize(cameraPosition - vW);
        float edge = pow(abs(dot(normalize(vN), V)), 1.7);
        float len = pow(clamp(vUv.y, 0.0, 1.0), 1.6);
        float tip = smoothstep(1.0, 0.97, vUv.y);

        float fogF = 1.0 - exp(-dist * 0.018);
        float a = uIntensity * edge * len * tip * (1.0 - fogF * 0.85);
        gl_FragColor = vec4(uColor, a);
      }`
