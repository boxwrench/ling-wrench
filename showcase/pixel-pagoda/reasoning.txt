```js
const FACES = [ ... ];
const vb = { pos: [], col: [], idx: [], n: 0 };
const eb = { pos: [], idx: [], n: 0 };
const tmpC = new THREE.Color();
function pushBox(b, minX, minY, minZ, w, h, d, triple) {
  for (let f = 0; f < 6; f++) {
    const face = FACES[f];
    if (triple) { tmpC.setHex(triple[f]); }
    for (let c = 0; c < 4; c++) {
      const cn = face.corners[c];
      b.pos.push(minX + cn[0]*w, minY + cn[1]*h, minZ + cn[2]*d);
      if (triple) b.col.push(tmpC.r, tmpC.g, tmpC.b);
    }
    const n = b.n;
    b.idx.push(n, n+1, n+2, n+2, n+1, n+3);
    b.n += 4;
  }
}
function vox(x, y, z, w, h, d, triple) { pushBox(vb, x, y, z, w, h, d, triple); }
function glowBox(x, y, z, w, h, d) { pushBox(eb, x, y, z, w, h, d, null); }
```

Loop:

```js
const clock = new THREE.Clock();
function frame() {
  requestAnimationFrame(frame);
  const dt = Math.min(clock.getDelta(), 0.05);
  const t = clock.elapsedTime;
  const k = 1 - Math.exp(-dt * 2.4);
  // swing mood
  cur.sky.lerp(tgtMode.sky, k); ... colors; numbers:
  cur.lantern += (tgtMode.lantern - cur.lantern) * k; etc.
  applyParams();
  if (!interacted && !RM) theta += dt * 0.05;
  // camera
  const sp = Math.sin(phi), cp = Math.cos(phi);
  camera.position.set(TARGET.x + dist*sp*Math.sin(theta), TARGET.y + dist*cp, TARGET.z + dist*sp*Math.cos(theta));
  camera.lookAt(TARGET);
  // life
  if (!RM) updatePetals(dt, t);
  updateGlows(t);
  waterMat.uniforms.uTime.value = RM ? 0 : t;
  starMat.uniforms.uTime.value = RM ? 0 : t;
  // two passes
  renderer.setRenderTarget(rt);
  renderer.render(scene, camera);
  renderer.setRenderTarget(null);
  renderer.render(blitScene, blitCam);
}
```

```js
let theta = 0.7, phi = 0.62, dist = 27, interacted = false;
const pointers = new Map();
let pinchDist = 0;
canvas.addEventListener('pointerdown', (e) => {
  interacted = true;
  canvas.setPointerCapture(e.pointerId);
  pointers.set(e.pointerId, { x: e.clientX, y: e.clientY });
  if (pointers.size === 2) {
    const p = [...pointers.values()];
    pinchDist = Math.hypot(p[0].x - p[1].x, p[0].y - p[1].y);
  }
});
canvas.addEventListener('pointermove', (e) => {
  const p = pointers.get(e.pointerId);
  if (!p) return;
  const dx = e.clientX - p.x, dy = e.clientY - p.y;
  p.x = e.clientX; p.y = e.clientY;
  if (pointers.size === 1) {
    theta += dx * 0.005;
    phi = Math.min(1.42, Math.max(0.18, phi + dy * 0.005));
  } else if (pointers.size === 2) {
    const q = [...pointers.values()];
    const d = Math.hypot(q[0].x - q[1].x, q[0].y - q[1].y);
    if (pinchDist > 0) dist = Math.min(44, Math.max(10, dist * (pinchDist / d)));
    pinchDist = d;
  }
});
const endPointer = (e) => { pointers.delete(e.pointerId); pinchDist = 0; };
canvas.addEventListener('pointerup', endPointer);
canvas.addEventListener('pointercancel', endPointer);
canvas.addEventListener('wheel', (e) => {
  interacted = true;
  dist = Math.min(44, Math.max(10, dist * Math.exp(e.deltaY * 0.0012)));
}, { passive: true });
```
